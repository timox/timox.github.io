Debug Live des Références

Ouvrez la console du navigateur (F12) et exécutez ce code dans le contexte de votre page Kanban :

// Chercher toutes les tâches avec des notes
const tasks = kanbanManager.currentRecords;
tasks.forEach(task => {
    if (task.notes) {
        try {
            const notesData = JSON.parse(task.notes);
            if (notesData.references) {
                console.log(`Task ${task.id} - ${task.titre}:`);
                console.log('References:', notesData.references);
                
                // Chercher la carte dans le DOM
                const card = document.querySelector(`[data-id="${task.id}"]`);
                if (card) {
                    const refContainer = card.querySelector('.references-container');
                    console.log('References container found:', !!refContainer);
                    if (refContainer) {
                        console.log('Container HTML:', refContainer.innerHTML);
                        console.log('Container styles:', window.getComputedStyle(refContainer));
                    }
                } else {
                    console.log('Card not found in DOM');
                }
            }
        } catch (e) {
            console.error('Error parsing notes for task', task.id, e);
        }
    }
});

// Vérifier si le CSS est chargé
const styles = document.styleSheets;
let foundRefStyles = false;
for (let sheet of styles) {
    try {
        for (let rule of sheet.cssRules) {
            if (rule.selectorText && rule.selectorText.includes('reference')) {
                console.log('Found reference style:', rule.selectorText);
                foundRefStyles = true;
            }
        }
    } catch (e) {
        // Ignorer les erreurs CORS
    }
}
console.log('Reference styles loaded:', foundRefStyles);
